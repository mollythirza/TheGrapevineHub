  <script>
    const viewport = document.getElementById('viewport');
    const canvas = document.getElementById('canvas');

    let scale = 1;
    let pointX = 0;
    let pointY = 0;
    let startX = 0;
    let startY = 0;
    let isDragging = false;

    // Touch gesture state
    let initialPinchDistance = null;
    let initialScale = 1;

    function setTransform() {
      canvas.style.transform = `translate(${pointX}px, ${pointY}px) scale(${scale})`;
    }

    // --- MOUSE DRAGGING ---
    viewport.addEventListener('mousedown', (e) => {
      if (e.target.closest('a')) return;
      e.preventDefault();
      startX = e.clientX - pointX;
      startY = e.clientY - pointY;
      isDragging = true;
    });

    window.addEventListener('mousemove', (e) => {
      if (!isDragging) return;
      e.preventDefault();
      pointX = e.clientX - startX;
      pointY = e.clientY - startY;
      setTransform();
    });

    window.addEventListener('mouseup', () => { isDragging = false; });

    // --- MOUSE WHEEL ZOOM ---
    viewport.addEventListener('wheel', (e) => {
      e.preventDefault();
      const xs = (e.clientX - pointX) / scale;
      const ys = (e.clientY - pointY) / scale;
      const delta = -e.deltaY;
      
      if (delta > 0) scale = Math.min(scale * 1.15, 4.5);
      else scale = Math.max(scale / 1.15, 0.6);
      
      pointX = e.clientX - xs * scale;
      pointY = e.clientY - ys * scale;
      setTransform();
    }, { passive: false });

    // --- NATIVE MOBILE TOUCH & PINCH-TO-ZOOM ---
    function getDistance(t1, t2) {
      const dx = t1.clientX - t2.clientX;
      const dy = t1.clientY - t2.clientY;
      return Math.sqrt(dx * dx + dy * dy);
    }

    viewport.addEventListener('touchstart', (e) => {
      if (e.touches.length === 1) {
        if (e.target.closest('a')) return;
        startX = e.touches[0].clientX - pointX;
        startY = e.touches[0].clientY - pointY;
        isDragging = true;
      } else if (e.touches.length === 2) {
        isDragging = false;
        initialPinchDistance = getDistance(e.touches[0], e.touches[1]);
        initialScale = scale;
      }
    }, { passive: true });

    viewport.addEventListener('touchmove', (e) => {
      if (e.touches.length === 1 && isDragging) {
        pointX = e.touches[0].clientX - startX;
        pointY = e.touches[0].clientY - startY;
        setTransform();
      } else if (e.touches.length === 2 && initialPinchDistance) {
        const currentDistance = getDistance(e.touches[0], e.touches[1]);
        const factor = currentDistance / initialPinchDistance;
        scale = Math.min(Math.max(initialScale * factor, 0.6), 4.5);
        setTransform();
      }
    }, { passive: true });

    viewport.addEventListener('touchend', (e) => {
      if (e.touches.length < 2) initialPinchDistance = null;
      if (e.touches.length === 0) isDragging = false;
    });

    // --- BUTTON CONTROLS ---
    document.getElementById('zoom-in').addEventListener('click', () => { scale = Math.min(scale * 1.3, 4.5); setTransform(); });
    document.getElementById('zoom-out').addEventListener('click', () => { scale = Math.max(scale / 1.3, 0.6); setTransform(); });
    document.getElementById('zoom-reset').addEventListener('click', () => { scale = 1; pointX = 0; pointY = 0; setTransform(); });
  </script>
